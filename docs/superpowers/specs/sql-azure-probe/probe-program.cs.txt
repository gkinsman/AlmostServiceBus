string Detail(Result r)
{
    if (!r.Created) return r.CreateError?["type"] + ": " + Trim(r.CreateError?["message"]);
    var parts = new List<string>();
    if (r.SendError is not null) parts.Add("SEND " + r.SendError["message"]);
    if (r.Copies.Count > 1) parts.Add($"copies={r.Copies.Count}");
    foreach (var c in r.Copies.Concat(r.DeadLettered))
        parts.Add(string.Join(", ", c.Where(kv => kv.Key != "caseId" && kv.Value is not null).Select(kv => $"{kv.Key}={kv.Value}")));
    return string.Join(" / ", parts);
}
string Trim(string? s) => s is null ? "" : Regex.Replace(s, @"\s*(TrackingId|Status:|ErrorCode:|Headers:).*", "", RegexOptions.Singleline);

Dictionary<string, string?> Err(Exception ex)
{
    var d = new Dictionary<string, string?> { ["type"] = ex.GetType().Name, ["message"] = Scrub(ex.Message) };
    if (ex is ServiceBusException sbe) d["reason"] = sbe.Reason.ToString();
    if (ex is Azure.RequestFailedException rfe) { d["status"] = rfe.Status.ToString(); d["errorCode"] = rfe.ErrorCode; }
    if (ex.InnerException is { } inner) d["inner"] = inner.GetType().Name + ": " + Scrub(inner.Message);
    return d;
}

Dictionary<string, string?> Dump(ServiceBusReceivedMessage m)
{
    var d = new SortedDictionary<string, string?>
    {
        ["$Subject"] = m.Subject, ["$To"] = m.To, ["$CorrelationId"] = m.CorrelationId, ["$MessageId"] = m.MessageId, ["$ReplyTo"] = m.ReplyTo,
        ["$ReplyToSessionId"] = m.ReplyToSessionId, ["$SessionId"] = m.SessionId, ["$ContentType"] = m.ContentType, ["$PartitionKey"] = m.PartitionKey,
        ["$TimeToLive"] = m.TimeToLive == TimeSpan.MaxValue ? null : m.TimeToLive.ToString(),
        ["$ScheduledEnqueueTime"] = m.ScheduledEnqueueTime == default ? null : m.ScheduledEnqueueTime.ToString("o"),
    };
    foreach (var (k, v) in m.ApplicationProperties) d[k] = v is null ? "<null>" : $"{v} ({v.GetType().Name})";
    return new Dictionary<string, string?>(d);
}

CreateRuleOptions BuildRule(Case c, int k)
{
    var rule = c.Rules[k];
    var guard = $"user.caseId = '{c.Id}'";
    RuleFilter filter;
    if (rule.Corr)
        filter = new CorrelationRuleFilter { ApplicationProperties = { ["caseId"] = c.Id } };
    else
    {
        var sql = new SqlRuleFilter(rule.Filter is null ? guard : $"{guard} AND ({rule.Filter})");
        foreach (var (pk, pv) in rule.FP ?? []) sql.Parameters[pk] = pv;
        filter = sql;
    }
    var opts = new CreateRuleOptions($"r{k}", filter);
    if (rule.Action is not null)
    {
        var action = new SqlRuleAction(rule.Action);
        foreach (var (pk, pv) in rule.AP ?? []) action.Parameters[pk] = pv;
        opts.Action = action;
    }
    return opts;
}

static string Describe(Rule r) => $"filter={r.Filter ?? (r.Corr ? "<correlation>" : "<guard only>")}"
    + (r.FP is { Count: > 0 } ? " params=" + string.Join(",", r.FP.Select(kv => $"{kv.Key}={kv.Value}")) : "")
    + (r.Action is null ? "" : $" action={r.Action}")
    + (r.AP is { Count: > 0 } ? " aparams=" + string.Join(",", r.AP.Select(kv => $"{kv.Key}={kv.Value}")) : "");

record Rule(string? Filter, Dictionary<string, object?>? FP = null, string? Action = null, Dictionary<string, object?>? AP = null, bool Corr = false);
record Case(string Id, Rule[] Rules, Dictionary<string, object?> User, Action<ServiceBusMessage>? Sys);
class Result
{
    public string Id { get; set; } = "";
    public List<string> Rules { get; set; } = [];
    public string Outcome { get; set; } = "";
    public bool Created { get; set; }
    public Dictionary<string, string?>? CreateError { get; set; }
    public Dictionary<string, string?>? SendError { get; set; }
    public List<Dictionary<string, string?>> Copies { get; set; } = [];
    public List<Dictionary<string, string?>> DeadLettered { get; set; } = [];
}
