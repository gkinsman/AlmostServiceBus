using System.Globalization;
using System.Text.Json;
using System.Text.RegularExpressions;
using AlmostServiceBus.Tests.Shared;
using Azure.Messaging.ServiceBus;

// Turns the probe's case definitions plus Azure's recorded results (results*.json) into the shared table.
static class SqlProbeExport
{
    static readonly CultureInfo Recorded = new("en-AU");

    public static void Run(List<Case> cases, HashSet<string> noDlq, string outPath)
    {
        var results = new Dictionary<string, JsonElement>();
        foreach (var file in new[] { "results.json", "results-z..json", "results-y..json" })
            foreach (var c in JsonDocument.Parse(File.ReadAllText(file)).RootElement.GetProperty("cases").EnumerateArray())
                results[c.GetProperty("Id").GetString()!] = c;

        var table = new List<SqlCompatCase>();
        foreach (var c in cases)
        {
            if (!results.TryGetValue(c.Id, out var r)) { Console.WriteLine("no result: " + c.Id); continue; }
            if (r.GetProperty("SendError").ValueKind != JsonValueKind.Null) { Console.WriteLine("send error, skipped: " + c.Id); continue; }

            var parts = c.Id.Split('.');
            var entry = new SqlCompatCase
            {
                Id = c.Id,
                Area = parts[0] == "z" && parts.Length > 2 ? parts[1] : parts[0],
                Verified = "azure",
                Subscription = new SqlCompatSubscription { DeadLetterOnFilterExceptions = !noDlq.Contains(c.Id) },
                Rules = c.Rules.Select(rule => new SqlCompatRule
                {
                    Filter = rule.Filter,
                    FilterParameters = Typed(rule.FP),
                    Action = rule.Action,
                    ActionParameters = Typed(rule.AP),
                    Correlation = rule.Corr ? true : null,
                }).ToList(),
                Message = BuildMessage(c),
            };

            var wildcards = c.Rules.Where(x => x.Action is not null)
                .SelectMany(x => Regex.Matches(x.Action!, @"SET\s+user\.(\w+)\s*=\s*newid\(\)", RegexOptions.IgnoreCase))
                .Select(m => m.Groups[1].Value).ToHashSet(StringComparer.OrdinalIgnoreCase);

            var outcome = r.GetProperty("Outcome").GetString()!;
            entry.Expected.Outcome = outcome;
            if (outcome == "Rejected")
            {
                var error = r.GetProperty("CreateError");
                var message = error.GetProperty("message").GetString()!;
                var tracking = Regex.Match(message, @"\s*TrackingId:", RegexOptions.Singleline);
                entry.Expected.Error = new SqlCompatError
                {
                    Type = error.GetProperty("type").GetString()!,
                    Message = tracking.Success ? message[..tracking.Index] : null,
                };
                if (!tracking.Success)
                    entry.Note = "Rejected by the SDK client before reaching the service.";
            }
            else
            {
                foreach (var copy in r.GetProperty("Copies").EnumerateArray())
                    entry.Expected.Copies.Add(new SqlCompatCopy { User = UserProps(copy, wildcards), Sys = SysProps(copy) });
                foreach (var dead in r.GetProperty("DeadLettered").EnumerateArray())
                    entry.Expected.DeadLettered.Add(new SqlCompatDeadLetter
                    {
                        Reason = dead.GetProperty("$reason").GetString(),
                        Description = dead.GetProperty("$description").GetString(),
                        User = UserProps(dead, wildcards),
                    });
            }

            table.Add(entry);
        }

        SqlCompatCases.Save(table, outPath);
        Console.WriteLine($"exported {table.Count} cases to {outPath}");
    }

    static Dictionary<string, TypedValue>? Typed(Dictionary<string, object?>? values) =>
        values?.ToDictionary(kv => kv.Key, kv => TypedValue.From(kv.Value));

    static SqlCompatMessage BuildMessage(Case c)
    {
        var m = new ServiceBusMessage("body") { MessageId = "m-" + c.Id };
        c.Sys?.Invoke(m);
        var sys = new Dictionary<string, string?>
        {
            ["MessageId"] = m.MessageId,
            ["Label"] = m.Subject, ["To"] = m.To, ["ReplyTo"] = m.ReplyTo, ["CorrelationId"] = m.CorrelationId,
            ["ContentType"] = m.ContentType, ["ReplyToSessionId"] = m.ReplyToSessionId, ["SessionId"] = m.SessionId,
            ["PartitionKey"] = m.PartitionKey,
            ["TimeToLive"] = m.TimeToLive == TimeSpan.MaxValue ? null : m.TimeToLive.ToString("c", CultureInfo.InvariantCulture),
        };
        return new SqlCompatMessage
        {
            Sys = sys.Where(kv => kv.Value is not null).ToDictionary(),
            User = c.User.ToDictionary(kv => kv.Key, kv => TypedValue.From(kv.Value)),
        };
    }

    static readonly HashSet<string> Skip = new(StringComparer.Ordinal)
        { "caseId", "DeadLetterReason", "DeadLetterErrorDescription", "DeadLetterSource" };

    static Dictionary<string, TypedValue> UserProps(JsonElement dump, HashSet<string> wildcards)
    {
        var user = new Dictionary<string, TypedValue>();
        foreach (var p in dump.EnumerateObject())
        {
            if (p.Name.StartsWith('$') || Skip.Contains(p.Name)) continue;
            var text = p.Value.GetString()!;
            var value = ParseDump(text);
            if (wildcards.Contains(p.Name) && value.Type != "null")
                value.Value = null; // any value of this type (newid)
            user[p.Name] = value;
        }
        return user;
    }

    static Dictionary<string, string?> SysProps(JsonElement dump)
    {
        var sys = new Dictionary<string, string?>();
        foreach (var p in dump.EnumerateObject())
            if (p.Name.StartsWith('$') && p.Name is not "$reason" and not "$description" && p.Value.ValueKind == JsonValueKind.String)
                sys[p.Name[1..]] = p.Value.GetString();
        return sys;
    }

    static TypedValue ParseDump(string text)
    {
        if (text == "<null>") return new TypedValue { Type = "null" };
        var i = text.LastIndexOf(" (", StringComparison.Ordinal);
        var raw = text[..i];
        var type = text[(i + 2)..^1];
        object value = type switch
        {
            "String" => raw,
            "Int32" => int.Parse(raw, CultureInfo.InvariantCulture),
            "Int64" => long.Parse(raw, CultureInfo.InvariantCulture),
            "UInt32" => uint.Parse(raw, CultureInfo.InvariantCulture),
            "UInt64" => ulong.Parse(raw, CultureInfo.InvariantCulture),
            "Int16" => short.Parse(raw, CultureInfo.InvariantCulture),
            "Byte" => byte.Parse(raw, CultureInfo.InvariantCulture),
            "SByte" => sbyte.Parse(raw, CultureInfo.InvariantCulture),
            "Char" => raw[0],
            "Single" => float.Parse(raw, Recorded),
            "Double" => double.Parse(raw, Recorded),
            "Decimal" => decimal.Parse(raw, Recorded),
            "Boolean" => bool.Parse(raw),
            "Guid" => Guid.Parse(raw),
            "TimeSpan" => TimeSpan.Parse(raw, CultureInfo.InvariantCulture),
            "DateTime" => DateTime.Parse(raw, Recorded, DateTimeStyles.AssumeUniversal | DateTimeStyles.AdjustToUniversal),
            "DateTimeOffset" => DateTimeOffset.Parse(raw, Recorded),
            "Uri" => new Uri(raw),
            _ => throw new InvalidOperationException("unknown dump type " + type)
        };
        return TypedValue.From(value);
    }
}
